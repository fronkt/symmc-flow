\documentclass[border=4pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta,positioning,fit,backgrounds,calc}

\begin{document}
\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.2mm]},
  mol/.style={regular polygon,regular polygon sides=5,draw,minimum size=6mm,inner sep=0pt,thick},
  box/.style={rounded corners,draw,thick,inner sep=6pt},
  lbl/.style={font=\footnotesize\itshape},
]

% ---------------------------------------------------------------------------
% Panel (a): factorization  cart_{m,a} = c_m L + R_m local_{m,a}
% ---------------------------------------------------------------------------
\node[lbl] at (-0.2,3.1) {(a) Rigid-body factorization};

% unit cell
\begin{scope}[shift={(0,0)}]
  \draw[thick] (0,0) -- (3,0) -- (3.7,1.2) -- (0.7,1.2) -- cycle;       % bottom face
  \draw[thick] (0,0) -- (0,2.4) (3,0) -- (3,2.4)
               (3.7,1.2) -- (3.7,3.6) (0.7,1.2) -- (0.7,3.6);           % verticals
  \draw[thick] (0,2.4) -- (3,2.4) -- (3.7,3.6) -- (0.7,3.6) -- cycle;   % top face
  \node[lbl] at (1.55,-0.35) {lattice $L\in\mathbb{R}^{3\times3}$};

  % two symmetry-related molecules (rigid pentagons) at fractional centroids
  \node[mol,rotate=12,fill=blue!12] (m1) at (1.15,1.25) {};
  \node[mol,rotate=-150,fill=red!12] (m2) at (2.35,2.05) {};
  \fill (m1.center) circle (0.7pt); \fill (m2.center) circle (0.7pt);
  \node[lbl,blue!55!black] at (0.65,1.7) {$c_1,R_1$};
  \node[lbl,red!55!black]  at (2.95,1.75) {$c_2,R_2$};
\end{scope}

\node[anchor=west] at (4.4,1.8)
  {$\displaystyle \text{cart}_{m,a}=c_m L + R_m\,\text{local}_{m,a}$};
\node[lbl,anchor=west] at (4.4,1.1)
  {centroid $c_m\in\mathbb{T}^3$,\; orientation $R_m\in\mathrm{SO}(3)$};

% ---------------------------------------------------------------------------
% Panel (b): orientation decomposition  R_m = rot(g_m) . R_asym
% ---------------------------------------------------------------------------
\node[lbl] at (-0.2,-1.0) {(b) Orientation decomposition};

\node[box,fill=black!4] (Rm) at (0.9,-2.2) {$R_m$};
\node[box,fill=green!10] (rg) at (4.0,-1.6)
  {$\mathrm{rot}(g_m)$};
\node[box,fill=orange!12] (ra) at (4.0,-2.8)
  {$R_{\text{asym}}$};

\draw[->,thick] (Rm) -- (rg.west) node[midway,above,sloped,font=\scriptsize]{};
\draw[->,thick] (Rm) -- (ra.west);
\node at (2.45,-2.2) {$=$};

\node[anchor=west,green!45!black,font=\footnotesize,align=left] at (5.7,-1.6)
  {space-group--determined\\relative rotation \,\textbf{(learnable)}};
\node[anchor=west,orange!55!black,font=\footnotesize,align=left] at (5.7,-2.8)
  {asymmetric-unit free pose\\gauge-arbitrary \,\textbf{(not learnable here)}};

\end{tikzpicture}
\end{document}
